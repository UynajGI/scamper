```rust
// CMC.rs/src/lattice/builders.rs

use super::{BondType, Lattice, Neighbor};

/// Build 1D chain lattice.
pub fn build_chain(n_sites: usize, pbc: bool) -> Lattice {
    let mut sites = vec![vec![]; n_sites];
    let mut n_bonds = 0;

    for i in 0..n_sites {
        let right = if pbc {
            (i + 1) % n_sites
        } else {
            i + 1
        };
        if right < n_sites {
            sites[i].push(Neighbor {
                target: right,
                bond_type: BondType::ChainX,
            });
            n_bonds += 1;
        }
    }

    Lattice {
        sites,
        n_sites,
        n_bonds,
    }
}

/// Build 2D square lattice.
pub fn build_square(lx: usize, ly: usize, pbc: bool) -> Lattice {
    let n_sites = lx * ly;
    let mut sites = vec![vec![]; n_sites];
    let mut n_bonds = 0;

    for y in 0..ly {
        for x in 0..lx {
            let idx = y * lx + x;

            // Right neighbor (X direction)
            let rx = if pbc { (x + 1) % lx } else { x + 1 };
            if rx < lx {
                let target = y * lx + rx;
                sites[idx].push(Neighbor {
                    target,
                    bond_type: BondType::SquareX,
                });
                n_bonds += 1;
            }

            // Down neighbor (Y direction)
            let ry = if pbc { (y + 1) % ly } else { y + 1 };
            if ry < ly {
                let target = ry * lx + x;
                sites[idx].push(Neighbor {
                    target,
                    bond_type: BondType::SquareY,
                });
                n_bonds += 1;
            }
        }
    }

    Lattice {
        sites,
        n_sites,
        n_bonds,
    }
}
```

```rust
// CMC.rs/src/models/ising.rs (continue)

impl ModelMC for IsingModel {
    fn spin_dim(&self) -> usize {
        1
    }

    fn coupling(&self) -> f64 {
        self.j
    }

    fn beta(&self) -> f64 {
        self.beta
    }

    fn local_energy_change(&self, site: usize, old: f64, new: f64) -> f64 {
        let mut delta_e = 0.0;
        let site_idx = site;
        let neighbors = &self.lattice.sites[site_idx];

        for neighbor in neighbors {
            let neighbor_spin = self.spins[neighbor.target];
            // H = -J Σ S_i S_j, ΔE = 2 * J * S_i_old * S_neighbor
            delta_e += 2.0 * self.j * old * neighbor_spin;
        }

        // Also account for reverse bonds (since adjacency list is directed)
        // Each bond appears once from each direction, so energy change is
        // computed correctly by summing over outgoing neighbors only.
        delta_e
    }

    fn total_energy(&self) -> f64 {
        let mut energy = 0.0;
        for (site_idx, neighbors) in self.lattice.sites.iter().enumerate() {
            for neighbor in neighbors {
                energy -= self.j * self.spins[site_idx] * self.spins[neighbor.target];
            }
        }
        // Each bond counted once (directed adjacency list)
        energy
    }

    fn spins(&self) -> &[f64] {
        &self.spins
    }

    fn spins_mut(&mut self) -> &mut [f64] {
        &mut self.spins
    }
}
```

```rust
// CMC.rs/src/algorithms/metropolis.rs (continue)

impl<MC: ModelMC> MonteCarlo for MetropolisCore<MC> {
    type Rng = Xoshiro256PlusPlus;

    fn sweep(&mut self, ctx: &mut Context<Self::Rng>) {
        let n = self.model.n_sites();
        for _ in 0..n {
            let site = ctx.rng.random_range(0..n);
            let old = self.model.spins()[site];
            let new = -old; // Ising flip: +1 → -1, -1 → +1
            let de = self.model.local_energy_change(site, old, new);
            if de < 0.0 || ctx.rng.random::<f64>() < (-self.model.beta() * de).exp() {
                self.model.spins_mut()[site] = new;
            }
        }
    }

    fn measure(&mut self, ctx: &mut Context<Self::Rng>) {
        let energy = self.model.total_energy();
        ctx.measure("Energy", energy);
    }

    fn name(&self) -> &'static str {
        "MetropolisCore"
    }
}
```

```rust
mod tests {
    use super::*;
    use crate::lattice::build_chain;
    use crate::models::IsingModel;

    #[test]
    fn test_metropolis_sweep() {
        let lattice = build_chain(8, true);
        let model = IsingModel::new(lattice, 1.0, 1.0);
        let mut core = MetropolisCore::new(model);
        let mut rng = Xoshiro256PlusPlus::seed_from_u64(42);
        let mut ctx = Context::new(rng);

        // After many sweeps, energy should decrease from all-up initial state
        let initial_energy = core.model.total_energy();

        for _ in 0..100 {
            core.sweep(&mut ctx);
        }

        let final_energy = core.model.total_energy();
        // At low temperature, energy should decrease (or stay similar)
        assert!(final_energy <= initial_energy + 1e-10);
    }

    #[test]
    fn test_metropolis_high_temp_disorder() {
        let lattice = build_chain(16, true);
        let model = IsingModel::new(lattice, 0.01, 1.0); // Very high temperature
        let mut core = MetropolisCore::new(model);
        let mut rng = Xoshiro256PlusPlus::seed_from_u64(42);
        let mut ctx = Context::new(rng);

        // At high temperature, most flips should be accepted
        let initial_spins = core.model.spins().to_vec();
        for _ in 0..1000 {
            core.sweep(&mut ctx);
        }
        let final_spins = core.model.spins().to_vec();

        // At high T, should have significant spin flips
        let n_flipped: usize = initial_spins
            .iter()
            .zip(final_spins.iter())
            .filter(|(a, b)| a != b)
            .count();
        // Expect many spins to have flipped at high temperature
        assert!(n_flipped > 0);
```

```rust
impl<MC: ModelMC> MonteCarlo for WolffCore<MC> {
    type Rng = Xoshiro256PlusPlus;

    fn sweep(&mut self, ctx: &mut Context<Self::Rng>) {
        let n = self.model.n_sites();
        let p_add = 1.0 - (-2.0 * self.model.coupling() * self.model.beta()).exp();

        // Wolff: grow one cluster and flip it
        let seed = ctx.rng.random_range(0..n);
        let seed_spin = self.model.spins()[seed];
        let mut cluster = Vec::new();
        let mut stack = VecDeque::new();

        cluster.push(seed);
        stack.push_back(seed);

        while let Some(site) = stack.pop_front() {
            for neighbor in &self.model.lattice().sites[site] {
                let nb_spin = self.model.spins()[neighbor.target];
                // Only add same-spin neighbors
                if nb_spin == seed_spin && !cluster.contains(&neighbor.target) {
                    if ctx.rng.random::<f64>() < p_add {
                        cluster.push(neighbor.target);
                        stack.push_back(neighbor.target);
                    }
                }
            }
        }

        // Flip entire cluster
        let new_spin = -seed_spin;
        for &site in &cluster {
            self.model.spins_mut()[site] = new_spin;
        }
    }

    fn measure(&mut self, ctx: &mut Context<Self::Rng>) {
        let energy = self.model.total_energy();
        ctx.measure("Energy", energy);
    }

    fn name(&self) -> &'static str {
        "WolffCore"
    }
}
```

```rust
use crate::models::ModelMC;
use crate::{CarloError, Context, FromParams, MonteCarlo, Params};
use rand::Rng;
use rand_xoshiro::Xoshiro256PlusPlus;

/// Union-Find data structure for cluster identification.
struct UnionFind {
    parent: Vec<usize>,
}

impl UnionFind {
    fn new(n: usize) -> Self {
        UnionFind {
            parent: (0..n).collect(),
        }
    }

    fn find(&mut self, x: usize) -> usize {
        if self.parent[x] != x {
            self.parent[x] = self.find(self.parent[x]);
        }
        self.parent[x]
    }

    fn union(&mut self, x: usize, y: usize) {
        let rx = self.find(x);
        let ry = self.find(y);
        if rx != ry {
            self.parent[rx] = ry;
        }
    }
}
```

```rust
// CMC.rs/src/algorithms/swendsen_wang.rs (continue)

impl<MC: ModelMC> MonteCarlo for SWCore<MC> {
    type Rng = Xoshiro256PlusPlus;

    fn sweep(&mut self, ctx: &mut Context<Self::Rng>) {
        let n = self.model.n_sites();
        let p_add = 1.0 - (-2.0 * self.model.coupling() * self.model.beta()).exp();

        // Step 1: Build FK clusters
        let mut uf = UnionFind::new(n);

        for site in 0..n {
            let site_spin = self.model.spins()[site];
            for neighbor in &self.model.lattice().sites[site] {
                let nb_spin = self.model.spins()[neighbor.target];
                if site_spin == nb_spin && ctx.rng.random::<f64>() < p_add {
                    uf.union(site, neighbor.target);
                }
            }
        }

        // Step 2: Assign random spin to each cluster
        let mut cluster_roots = std::collections::HashSet::new();
        for i in 0..n {
            cluster_roots.insert(uf.find(i));
        }

        let cluster_spins: std::collections::HashMap<usize, f64> = cluster_roots
            .into_iter()
            .map(|root| {
                let spin = if ctx.rng.random::<f64>() < 0.5 { 1.0 } else { -1.0 };
                (root, spin)
            })
            .collect();

        // Step 3: Apply cluster spins
        for i in 0..n {
            let root = uf.find(i);
            self.model.spins_mut()[i] = cluster_spins[&root];
        }
    }

    fn measure(&mut self, ctx: &mut Context<Self::Rng>) {
        let energy = self.model.total_energy();
        ctx.measure("Energy", energy);
    }

    fn name(&self) -> &'static str {
        "SWCore"
    }
}
```
